```js
// lion-hawk phase 1 (G-2): every workspace resolves the same three and Rapier, and the engine is a real folder
// inside games/ so its bare imports resolve from this workspace, not from wherever a symlink points.
import test from 'node:test';
import assert from 'node:assert/strict';
import { createRequire } from 'node:module';
import { realpathSync, readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';

const games = realpathSync(fileURLToPath(new URL('..', import.meta.url)));
const members = ['lion-hawk-engine', 'pig-pens', 'tractor-pickup'];
const resolveFrom = (dir, spec) => realpathSync(createRequire(`${games}/${dir}/package.json`).resolve(spec));

test('the engine is a real folder under games/', () => {
  assert.equal(realpathSync(`${games}/lion-hawk-engine`), `${games}/lion-hawk-engine`);
  assert.ok(resolveFrom('pig-pens', 'lion-hawk/package.json').startsWith(`${games}/lion-hawk-engine/`));
});
for (const spec of ['three/package.json', '@dimforge/rapier3d-compat/package.json']) {
  test(`one copy of ${spec.split('/package.json')[0]} across the workspace`, () => {
    const paths = new Set(members.map(m => resolveFrom(m, spec)));
    assert.equal(paths.size, 1, [...paths].join('\n'));
  });
}
test('the shared copies satisfy the engine peer ranges', () => {
  const peers = JSON.parse(readFileSync(`${games}/lion-hawk-engine/package.json`, 'utf8')).peerDependencies;
  assert.deepEqual(Object.keys(peers).sort(), ['@dimforge/rapier3d-compat', 'three']);
  const version = spec => JSON.parse(readFileSync(resolveFrom('tractor-pickup', `${spec}/package.json`), 'utf8')).version;
  assert.match(version('three'), /^0\.186\./);
  assert.match(version('@dimforge/rapier3d-compat'), /^0\.21\./);
});
```

- [ ] **Step 3: Add the Tractor Pickup pickup scenario**

Append to `games/tractor-pickup/test/golden.test.mjs`. It uses `createGame`, imported above:
```js
// The drive above never picks anything up; this one boops three animals into the trailer and drives on, so the
// pickup → flight → land → ride loop is covered (phase 1 carry-over).
const STILL = { thr: 0, steer: 0, horn: false };
function pickup() {
  const g = createGame(RAPIER, { seed: 14, power: 'medium' }), events = [], trail = [];
  const step = input => { for (const e of g.step(input)) events.push(`${trail.length}:${e.type}`); };
  g.herd.animals.filter(a => a.home === 'yard').forEach(a => { a.state = 'gone'; });          // only placed animals get booped
  for (let i = 0; i < 60; i++) step(STILL);
  const single = g.herd.free().filter(x => !x.hidden && x.type !== 'chick' && !g.herd.animals.some(c => c.leader === x.id));
  for (const a of single.slice(0, 3)) {
    const p = g.tractorWorld({ x: 2.5, y: 0, z: 0 }, {}); a.x = p.x; a.z = p.z; a.state = 'idle'; a.timer = 99;
    for (let i = 0; i < 90; i++) step(STILL);
  }
  for (let i = 0; i < 600; i++) { step({ thr: 1, steer: Math.sin(i / 50) * 0.5, horn: false }); if (i % 60 === 59) trail.push(checksum(snapshot(g))); }
  return { state: { ...snapshot(g), slots: g.load.slots.map(s => ({ car: s.car, animal: s.animal?.id ?? null })) }, events, trail };
}
test('golden: seed 14 pickup of three animals, then a 10 s drive', () => {
  const r = pickup();
  assert.equal(r.state.landed, 3, 'three animals ride');
  assert.ok(r.events.some(e => e.endsWith(':land')));
  golden(new URL('./goldens/pickup-seed14.json', import.meta.url), r);
});
```

If `landed` is not 3, read `test/game.test.mjs` (`place`, `pickable`) and adapt the placement so that it matches. Do not weaken the assertion.
