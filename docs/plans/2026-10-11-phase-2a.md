### Task 1: Schedule (`src/core/schedule.js`)

**Interfaces:**
- Produces:
  - `STEP_STAGES`: the frozen stage list above.
  - `createSchedule(stages: string[])`, which returns:
    - `add(name, fn, { stage, before = [], after = [] }) → remove()`;
    - `remove(name)`;
    - `order(stage) → string[]`;
    - `run(stage, ...args)`, which calls each fn in order with `args`;
    - `describe() → { [stage]: string[] }`.
  - Errors:
    - an unknown stage throws `schedule: unknown stage '<stage>' for '<name>'`;
    - a duplicate name throws `schedule: duplicate system '<name>'`;
    - a cycle throws `schedule: cycle in '<stage>': <names, comma-separated>`.
  - `before`/`after` names that are missing, or that live in another stage, are ignored. A kit that isn't loaded is simply not there.
  - Ties are broken by insertion order, so the result is stable. The order is cached per stage until the next `add`/`remove`.

```js
// R-6: the step runs named stages in a fixed order; systems inside a stage order themselves with before/after
// (the Threlte model). Unknown references are ignored so a game can name a kit's system that is not loaded.
export const STEP_STAGES = Object.freeze(['input.apply', 'pre', 'physics.world', 'post', 'events.flush']);

export function createSchedule(stages) {
  const byStage = new Map(stages.map(s => [s, []])), byName = new Map(), sorted = new Map();
  let n = 0;
  function sort(stage) {
    const list = byStage.get(stage), names = new Set(list.map(e => e.name)), deps = new Map(list.map(e => [e.name, new Set()]));
    for (const e of list) {
      for (const a of e.after) if (names.has(a)) deps.get(e.name).add(a);
      for (const b of e.before) if (names.has(b)) deps.get(b).add(e.name);
    }
    const out = [], done = new Set();
    while (out.length < list.length) {
      const next = list.find(e => !done.has(e.name) && [...deps.get(e.name)].every(d => done.has(d))); // lowest insertion index first
      if (!next) throw new Error(`schedule: cycle in '${stage}': ${list.filter(e => !done.has(e.name)).map(e => e.name).join(', ')}`);
      done.add(next.name); out.push(next);
    }
    return out;
  }
  const entries = stage => { if (!sorted.has(stage)) sorted.set(stage, sort(stage)); return sorted.get(stage); };
  const api = {
    add(name, fn, { stage, before = [], after = [] } = {}) {
      if (!byStage.has(stage)) throw new Error(`schedule: unknown stage '${stage}' for '${name}'`);
      if (byName.has(name)) throw new Error(`schedule: duplicate system '${name}'`);
      const e = { name, fn, stage, before: [...before], after: [...after], i: n++ };
      byName.set(name, e); byStage.get(stage).push(e); sorted.delete(stage);
      return () => api.remove(name);
    },
    remove(name) {
      const e = byName.get(name); if (!e) return;
      byName.delete(name); const list = byStage.get(e.stage); list.splice(list.indexOf(e), 1); sorted.delete(e.stage);
    },
    order: stage => entries(stage).map(e => e.name),
    run(stage, ...args) { const list = entries(stage); for (let i = 0; i < list.length; i++) list[i].fn(...args); },
    describe: () => Object.fromEntries([...byStage.keys()].map(s => [s, api.order(s)])),
  };
  return api;
}
```

### Task 2: Loop (`src/core/loop.js`)

**Interfaces:**
- Produces `createLoop({ hz = 60, maxCatchUp = 3, maxFrame = 0.25, step, render = () => {} })`, which returns:
  - `dt`: `1/hz`;
  - `alpha` (getter): `acc/dt`;
  - `stats`: `{ frames, steps, clamped, lastSteps }`, mutated in place;
  - `frame(seconds) → steps run`:
    1. clamps `seconds` to `[0, maxFrame]` and adds it to the accumulator;
    2. runs up to `maxCatchUp` steps;
    3. if a whole step or more is still left, counts `clamped++` and keeps only the fractional remainder ([R-2]);
    4. calls `render(alpha)` ([R-3]).
  - `stepN(n)`: exactly `n` steps, no render;
  - `advance(seconds) → steps run`: accumulates with no cap and keeps the remainder, no render ([R-7]);
  - `reset()`: clears the accumulator, for when the tab becomes visible again ([R-5]).
- `step(dt)` is always called with the fixed `dt`. Accumulator comparisons use an epsilon of `1e-9`, so that `advance(1)` at 60 Hz runs exactly 60 steps.

`src/core/loop.js`:
```js
// R-1…R-3, R-7: fixed steps from an accumulator, a catch-up cap that drops (and counts) excess time, and an alpha for
// render interpolation. No rAF here: the view's driver calls frame(seconds); tests call stepN/advance.
const EPS = 1e-9;
export function createLoop({ hz = 60, maxCatchUp = 3, maxFrame = 0.25, step, render = () => {} }) {
  const dt = 1 / hz, stats = { frames: 0, steps: 0, clamped: 0, lastSteps: 0 };
  let acc = 0;
  const run = cap => { let n = 0; while (acc >= dt - EPS && n < cap) { step(dt); acc -= dt; n++; } if (acc < 0) acc = 0; stats.steps += n; return n; };
  return {
    dt, stats,
    get alpha() { return acc / dt; },
    frame(seconds) {
      acc += Math.min(Math.max(seconds, 0), maxFrame);
      const n = run(maxCatchUp);
      if (acc >= dt - EPS) { stats.clamped++; acc %= dt; } // R-2: drop what the cap could not run
      stats.frames++; stats.lastSteps = n; render(acc / dt);
      return n;
    },
    stepN(n) { for (let i = 0; i < n; i++) step(dt); stats.steps += n; },
    advance(seconds) { acc += Math.max(seconds, 0); return run(Infinity); },
    reset() { acc = 0; },
  };
}
```

### Task 3: Input actions (`src/core/actions.js`)

**Interfaces:**
- Produces:
  - `axis()`, `button()`: frozen descriptors `{ kind: 'axis' | 'button' }`;
  - `actions(map) → frozen map`. A value that isn't one of those descriptors throws `actions: '<name>' must be axis() or button()`.
  - `createInput(schema)`, which returns:
    - `snapshot`: a live object. Each axis holds a number in [-1, 1]; each button holds `{ down, pressed, released }`.
    - `device(partial)`: device layer, for keyboard, touch and gamepad drivers; sticky until overwritten.
    - `set(partial)`: applies to the next `apply()` only ([I-6]).
    - `hold(partial)`: sticky override.
    - `release(...names)`: with no names, clears every hold.
    - `apply() → snapshot`: once per step ([I-4]). It computes axes and button edges and clears the `set` layer.
    - `values() → plain { name: number | boolean }`: a new object, for replays and replication.
  - Partial values: axes take numbers, buttons take booleans.
  - Precedence: hold > set > device.
  - An unknown action name throws `input: unknown action '<name>'`.
  - `apply()` allocates nothing.

```js
// I-1, I-4, I-6: games declare named actions; devices, tests and the network write layered values; once per step
// apply() turns them into one snapshot that systems read. Axes are numbers in [-1, 1]; buttons carry edges.
export const axis = () => Object.freeze({ kind: 'axis' });
export const button = () => Object.freeze({ kind: 'button' });
export function actions(map) {
  for (const [k, v] of Object.entries(map)) if (v?.kind !== 'axis' && v?.kind !== 'button') throw new Error(`actions: '${k}' must be axis() or button()`);
  return Object.freeze({ ...map });
}

export function createInput(schema) {
  const names = Object.keys(schema), isAxis = Object.fromEntries(names.map(n => [n, schema[n].kind === 'axis']));
  const snapshot = Object.fromEntries(names.map(n => [n, isAxis[n] ? 0 : { down: false, pressed: false, released: false }]));
  const device = {}, once = {}, held = {};
  const check = p => { for (const k in p) if (!(k in isAxis)) throw new Error(`input: unknown action '${k}'`); return p; };
  const clear = o => { for (const k in o) delete o[k]; };
  return {
    snapshot,
    device: p => { Object.assign(device, check(p)); },
    set: p => { Object.assign(once, check(p)); },
    hold: p => { Object.assign(held, check(p)); },
    release(...ns) { if (!ns.length) clear(held); else for (const n of ns) delete held[n]; },
    apply() {
      for (let i = 0; i < names.length; i++) {
        const n = names[i], v = n in held ? held[n] : n in once ? once[n] : device[n];
        if (isAxis[n]) snapshot[n] = typeof v === 'number' && v === v ? Math.max(-1, Math.min(1, v)) : 0;
        else { const b = snapshot[n], d = !!v; b.pressed = d && !b.down; b.released = !d && b.down; b.down = d; }
      }
      clear(once); return snapshot;
    },
    values: () => Object.fromEntries(names.map(n => [n, isAxis[n] ? snapshot[n] : snapshot[n].down])),
  };
}
```

### Task 4: UI registry (`src/core/ui.js`)

**Interfaces:**
- Produces `createUi({ onAction })`, which returns:
  - `builder(scene) → { button({ id, label, action, when }), toggle({ id, label, get, set, when }) }`.
    - Registers controls under the test ID `<scene>.<id>`. A duplicate test ID throws `ui: duplicate control '<testid>'`.
  - `clear(scene)`: removes that scene's controls.
  - `tree(ctx) → [{ id, scene, role, label, enabled, pressed? }]`, in registration order:
    - `role` is `'button'` or `'switch'`;
    - `enabled` is `when ? !!when(ctx) : true`;
    - `pressed` is present only on toggles, as `!!get(ctx)`.
  - `tap(testid, ctx)`:
    - throws `ui: no control '<testid>'` if the ID is unknown, and `ui: control '<testid>' is disabled` if `when` is false;
    - for a button: a function `action` is called with `ctx`; a string `action` calls `onAction(action)`, which presses that input button for one step;
    - for a toggle: calls `set(!get(ctx), ctx)`.
- `ctx` is whatever the engine passes (the engine itself).

`src/core/ui.js`:
```js
// U-1: scenes declare their controls; the browser renders them as real buttons with data-testid="<scene>.<id>",
// and headless tests tap them through this registry (T-5).
export function createUi({ onAction }) {
  const controls = new Map();
  const add = (scene, role, spec) => {
    const id = `${scene}.${spec.id}`;
    if (controls.has(id)) throw new Error(`ui: duplicate control '${id}'`);
    controls.set(id, { ...spec, id, scene, role });
  };
  return {
    builder: scene => ({ button: spec => add(scene, 'button', spec), toggle: spec => add(scene, 'switch', spec) }),
    clear(scene) { for (const [id, c] of controls) if (c.scene === scene) controls.delete(id); },
    tree: ctx => [...controls.values()].map(c => {
      const o = { id: c.id, scene: c.scene, role: c.role, label: c.label, enabled: c.when ? !!c.when(ctx) : true };
      if (c.role === 'switch') o.pressed = !!c.get(ctx);
```

**Interfaces:**
- Consumes Tasks 1–4, plus `rng` from `src/core/rng.js` and `createEvents` from `src/core/events.js`.
- `defineGame(spec) → frozen game`:
  - Required:
    - `id`: a non-empty string;
    - `scenes`: an object with at least one entry;
    - `start`: a key of `scenes`.
  - Optional:
    - `sim: { hz = 60, maxCatchUp = 3 }`;
    - `actions`, from `actions()` (default `actions({})`);
    - `policies` (default `{}`);
    - `state(engine) → object`;
    - `debug(engine) → object`.
  - Each scene may have `enter(engine, ctx)`, `exit(engine, ctx)`, `step(engine, dt, ctx)`, `ui(builder, engine, ctx)`, `presets: { name(engine, opts, ctx) }` and `pauses: boolean`.
  - Problems throw `defineGame: <reason>`. Use these exact reasons:
    - `id must be a non-empty string`
    - `scenes must not be empty`
    - `start '<x>' is not a scene`
    - `scene '<name>'.<hook> must be a function`
- `createEngine(game, { seed = 1, scene = game.start, args } = {}) → engine` (sim-pure), where:
  - `ctx` is `{ name, args, state: {} }`, one per scene instance, passed to every hook.
  - Read-only properties:
    - `game`;
    - `seed`: the uint32 base seed;
    - `events`: a `createEvents` with `now = current step`;
    - `input`: `createInput(game.actions)`;
    - `time`: a getter returning `{ step, sim: step * dt }`;
    - `scene`: a getter returning `{ name, args, state, overlays: [names] }` for the main scene.
  - `rng(name)`: `rng(seed).fork(name)`.
  - `emit(type, data)`.
  - `system(name, fn, { stage = 'pre', before, after }) → remove`. `fn(engine, dt)`.
  - `schedule() → describe()`.
  - Scene control:
    - `go(name, args)`: exits overlays top-down, exits the main scene, enters the new main scene, emits `scene {name, from}`;
    - `overlay(name, args)`: enters the scene, pushes it, emits `overlay {name, open: true}`;
    - `close(name)`: exits that overlay, emits `overlay {name, open: false}`;
    - `jump(scene, preset, opts)`: calls `go(scene)`, then the preset. An unknown preset throws `engine: no preset '<preset>' in '<scene>'`.
    - An unknown scene name throws `engine: no scene '<name>'`.
  - Driving the loop: `stepN(n)`, `advance(seconds)`, `frame(seconds)` (driver), `onRender(fn)` (`fn(alpha)`, used by `frame`), `loop` (the `createLoop` instance).
  - UI: `ui` is `{ tree(), tap(testid) }`, bound to the engine. A string action presses that input button through `input.set({ [a]: true })`.
  - `state() → { scene: name, overlays, step, ...game.state?.(engine) }`.
- **Step pipeline** ([R-6]):
  1. run `'input.apply'` (the engine registers system `'input'` → `input.apply()`);
  2. if no open overlay has `pauses`:
     - run the main scene's `step`, then every overlay's `step`, as engine system `'scene'`, the first in `'pre'`;
     - run `'pre'`, `'physics.world'` and `'post'`;
  3. if an overlay pauses: run only the overlays' `step` hooks;
  4. run `'events.flush'` (system `'events'`);
  5. then increment `step`.

  Events emitted during step *k* carry `step: k`.
- When a scene is entered, its `ui` hook is called with `ui.builder(name)`. When it exits, `ui.clear(name)` is called. `createEngine` enters the start scene immediately. The `scene` event from `go` is flushed at the next step.

```js
// The engine core (spec §5): scenes with sim halves (S-1…S-5), the step pipeline over named stages (R-6), input
// applied once per step (I-4), the UI registry (U-1), events, seeded rng streams and the fixed-step loop (R-1…R-7).
// Sim-pure: the browser driver, devices and DOM rendering live in the view half (phase 2b).
import { rng } from './rng.js';
import { createEvents } from './events.js';
import { createSchedule, STEP_STAGES } from './schedule.js';
import { createLoop } from './loop.js';
import { actions as defineActions, createInput } from './actions.js';
import { createUi } from './ui.js';

const HOOKS = ['enter', 'exit', 'step', 'ui'];
export function defineGame(spec) {
  const fail = m => { throw new Error(`defineGame: ${m}`); };
  if (typeof spec.id !== 'string' || !spec.id) fail('id must be a non-empty string');
  if (!spec.scenes || !Object.keys(spec.scenes).length) fail('scenes must not be empty');
  if (!(spec.start in spec.scenes)) fail(`start '${spec.start}' is not a scene`);
  for (const [name, s] of Object.entries(spec.scenes)) for (const h of HOOKS) if (h in s && typeof s[h] !== 'function') fail(`scene '${name}'.${h} must be a function`);
  return Object.freeze({ ...spec, sim: { hz: 60, maxCatchUp: 3, ...spec.sim }, actions: spec.actions ?? defineActions({}), policies: spec.policies ?? {} });
}

export function createEngine(game, { seed = 1, scene = game.start, args } = {}) {
  const base = rng(seed), schedule = createSchedule(STEP_STAGES), input = createInput(game.actions);
  let step = 0, render = () => {};
  const events = createEvents({ now: () => step });
  const stack = []; // stack[0] is the main scene; the rest are overlays, bottom to top
  const def = name => game.scenes[name] ?? (() => { throw new Error(`engine: no scene '${name}'`); })();
  const ui = createUi({ onAction: a => input.set({ [a]: true }) });
  const enter = (name, a) => { const s = def(name), ctx = { name, args: a, state: {} }; stack.push({ s, ctx }); s.enter?.(engine, ctx); s.ui?.(ui.builder(name), engine, ctx); return ctx; };
  const exit = i => { const [{ s, ctx }] = stack.splice(i, 1); s.exit?.(engine, ctx); ui.clear(ctx.name); };
  const paused = () => { for (let i = 1; i < stack.length; i++) if (stack[i].s.pauses) return true; return false; };
  const stepScenes = (dt, from) => { for (let i = from; i < stack.length; i++) stack[i].s.step?.(engine, dt, stack[i].ctx); };
  const loop = createLoop({ hz: game.sim.hz, maxCatchUp: game.sim.maxCatchUp, render: a => render(a), step(dt) {
    schedule.run('input.apply', engine, dt);
    if (paused()) stepScenes(dt, 1);
    else { schedule.run('pre', engine, dt); schedule.run('physics.world', engine, dt); schedule.run('post', engine, dt); }
    schedule.run('events.flush', engine, dt);
    step++;
  } });
  const engine = {
    game, seed: base.seed, events, input, loop,
    get time() { return { step, sim: step / game.sim.hz }; }, // divide, don't multiply by dt: 180 * (1/60) is not exactly 3
    get scene() { const m = stack[0]?.ctx; return { name: m?.name, args: m?.args, state: m?.state ?? {}, overlays: stack.slice(1).map(o => o.ctx.name) }; },
    rng: name => base.fork(name),
    emit: (type, data) => events.emit(type, data),
    system: (name, fn, opts = {}) => schedule.add(name, fn, { stage: 'pre', ...opts }),
    schedule: () => schedule.describe(),
    go(name, a) { def(name); const from = stack[0]?.ctx.name; while (stack.length) exit(stack.length - 1); enter(name, a); events.emit('scene', { name, from }); return engine; },
    overlay(name, a) { enter(name, a); events.emit('overlay', { name, open: true }); return engine; },
    close(name) { const i = stack.findIndex((o, k) => k > 0 && o.ctx.name === name); if (i > 0) { exit(i); events.emit('overlay', { name, open: false }); } return engine; },
    jump(name, preset, opts) {
      const p = def(name).presets?.[preset];
      if (!p) throw new Error(`engine: no preset '${preset}' in '${name}'`);
      engine.go(name); p(engine, opts, stack[0].ctx); return engine;
    },
    stepN: n => loop.stepN(n), advance: s => loop.advance(s), frame: s => loop.frame(s),
    onRender(fn) { render = fn; },
    ui: { tree: () => ui.tree(engine), tap: id => ui.tap(id, engine) },
    state: () => ({ scene: stack[0]?.ctx.name, overlays: stack.slice(1).map(o => o.ctx.name), step, ...(game.state?.(engine) ?? {}) }),
  };
  schedule.add('input', () => input.apply(), { stage: 'input.apply' });
  schedule.add('scene', (e, dt) => stepScenes(dt, 0), { stage: 'pre' });
  schedule.add('events', () => events.flush(), { stage: 'events.flush' });
  enter(scene, args);
  return engine;
}
```

    - `untilScene(name, opts)`: true when the main scene or an overlay is `name`.

```js
// T-1…T-6: drive a game headless, in steps, with input policies, waits that explain their failures, event lists,
// scene jumps and UI taps. Everything counts steps, never milliseconds.
import { createEngine } from '../src/core/engine.js';
import { checksum } from './state.js';

export const policy = {
  still: () => ({}),
  const: p => () => p,
  seq: list => (h, i) => { let k = i; for (const [n, p] of list) { if (k < n) return p; k -= n; } return {}; },
  fn: f => f,
};
const matcher = m => typeof m === 'function' ? m : e => e.type === m;
export function eventList(arr) {
  return Object.assign(arr, {
    of: t => arr.filter(e => e.type === t),
    first: m => arr.find(matcher(m)),
    count: m => arr.filter(matcher(m)).length,
    types: () => arr.map(e => e.type),
  });
}

export function headless(game, { seed = 1, scene, args, invariants = [] } = {}) {
  const engine = createEngine(game, { seed, scene, args }), hz = game.sim.hz;
  const resolve = input => {
    if (input == null) return null;
    if (typeof input === 'function') return input;
    if (typeof input === 'string') { const p = game.policies[input]; if (!p) throw new Error(`headless: no policy '${input}'`); return p; }
    return () => input;
  };
  const one = (f, k) => { if (f) engine.input.set(f(h, k)); engine.stepN(1); for (const inv of invariants) inv(h); };
  const fmt = e => { const { seq, step, type, ...rest } = e; return `  @${step} ${type} ${JSON.stringify(rest)}`; };
  const h = {
    engine, seed: engine.seed,
    get scene() { return engine.scene; }, get time() { return engine.time; },
    // a policy's i counts from the start of each step()/until() call, so policy.seq schedules are relative to that call
    step(n = 1, input) { const f = resolve(input); for (let k = 0; k < n; k++) one(f, k); return h; },
    seconds: (s, input) => h.step(Math.round(s * hz), input),
    hold(p) { engine.input.hold(p); return h; },
    release(...ns) { engine.input.release(...ns); return h; },
    until(pred, { maxSteps, maxSeconds = 10, input, message } = {}) {
      const cap = maxSteps ?? Math.round(maxSeconds * hz), mark = engine.events.seq, f = resolve(input);
      for (let k = 0; k <= cap; k++) {
        if (pred(h)) return { steps: k, events: eventList(engine.events.since(mark)) };
        if (k < cap) one(f, k);
      }
      const last = engine.events.since(0).slice(-20).map(fmt).join('\n');
      throw new Error(`${message ?? 'until: condition not met'} after ${cap} steps (seed ${engine.seed}, step ${engine.time.step}, scene ${engine.scene.name})\nlast events:\n${last}`);
    },
    untilEvent(m, opts = {}) {
      const mark = engine.events.seq, f = matcher(m); let found;
      h.until(() => (found = engine.events.since(mark).find(f)) !== undefined, { message: `untilEvent(${typeof m === 'string' ? m : 'fn'})`, ...opts });
      return found;
    },
    untilScene(name, opts = {}) { return h.until(x => x.scene.name === name || x.scene.overlays.includes(name), { message: `untilScene(${name})`, ...opts }); },
    mark: () => engine.events.seq,
    since: m => eventList(engine.events.since(m)),
    tap(id) { engine.ui.tap(id); return h; },
    go(n, a) { engine.go(n, a); return h; },
    overlay(n, a) { engine.overlay(n, a); return h; },
    close(n) { engine.close(n); return h; },
    jump(s, p, o) { engine.jump(s, p, o); return h; },
    state: () => engine.state(),
    checksum: () => checksum(engine.state()),
  };
  return h;
}
```
